# --- survival: the two-library split locked by issue #3 ---
# torchsurv supplies the differentiable stratified Efron Cox loss that trains
# the encoder (stage one, and the end-to-end control).
torchsurv==0.2.0
torchmetrics==1.9.0

# lifelines.CoxPHFitter(penalizer=L, l1_ratio=0, strata=['study']) is the single
# decoder every model is refitted and scored through, and supplies
# proportional_hazard_test (Schoenfeld residuals). It is also what caps pandas
# at <3.0.
#
# The split is only safe because both libraries optimise the *same* objective:
# torchsurv's stratified Efron partial log-likelihood agrees with lifelines' to
# ~4e-5 (float32 rounding), while Breslow differs by over a log-unit. The
# agreement is specific to Efron, which is why Efron is locked and why
# tests/test_stack.py pins it.
lifelines==0.30.3

# scikit-survival supplies the metrics -- Uno's IPCW C-index and the integrated
# Brier score -- plus Random Survival Forest, which is viable only as an
# *additional* model.
#
# Its CoxnetSurvivalAnalysis is ruled out as the decoder: it has no `strata`
# parameter, so adopting it would silently reverse the stratification decision
# and let the cancer-type confound back into the loss.
#
# Its concordance index agrees with torchsurv's to 6 decimals on identical risk
# scores. lifelines.utils.concordance_index does NOT: it takes a predicted
# *time*, not a risk, so it returns 1 - C and reads as a bad model rather than
# raising. It is banned in model code (#3 §6); tests pin the trap.
scikit-survival==0.28.0
scikit-learn==1.9.0
